Usage

// Fetch agent registry
fetch('https://ai-sam-ai.github.io/agents/index.json')
  .then(r => r.json())
  .then(registry => {
    // Filter by segment
    const devAgents = registry.agents.filter(a => a.segment === 'software-development');

    // Or find customer-facing agents
    const publicAgents = registry.agents.filter(a => a.customer_facing);

    console.log(devAgents);
  });
